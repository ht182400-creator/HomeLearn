## 2. 表结构设计

### 2.1 用户与认证

#### User (家长用户)

```prisma
model User {
  id            String        @id @default(cuid())
  email         String        @unique
  username      String        @unique
  passwordHash  String
  role          UserRole      @default(PARENT)
  avatar        String?
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt
  
  // 关联
  children      ChildAccount[]
  questions     Question[]
  speechTemplates SpeechTemplate[]
  recordings    Recording[]
  
  @@map("users")
}

enum UserRole {
  PARENT      // 家长
  ADMIN       // 系统管理员(预留)
}
```

#### ChildAccount (孩子账户)

```prisma
model ChildAccount {
  id            String        @id @default(cuid())
  userId        String        // 所属家长
  nickname      String
  grade         Grade?        // 年级
  avatar        String?
  preferences   Json?         // 学习偏好设置
  isActive      Boolean       @default(true)
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt
  
  // 关联
  user          User          @relation(fields: [userId], references: [id], onDelete: Cascade)
  wrongQuestions WrongQuestion[]
  speechResults  SpeechResult[]
  reviewRecords  ReviewRecord[]
  
  @@index([userId])
  @@map("child_accounts")
}

enum Grade {
  PRIMARY_1    // 小学一年级
  PRIMARY_2
  PRIMARY_3
  PRIMARY_4
  PRIMARY_5
  PRIMARY_6
  MIDDLE_1     // 初一
  MIDDLE_2
  MIDDLE_3
  HIGH_1       // 高一
  HIGH_2
  HIGH_3
}
```

### 2.2 学科与知识点

#### Subject (学科)

```prisma
model Subject {
  id            String        @id @default(cuid())
  name          String        // 学科名称: 数学、英语、语文...
  code          String        @unique  // 代码: MATH, ENGLISH, CHINESE...
  icon          String?       // 图标URL
  color         String?       // 主题色
  order         Int          @default(0)  // 排序
  
  // 关联
  questions     Question[]
  topics        Topic[]
  
  @@map("subjects")
}
```

#### Topic (知识点)

```prisma
model Topic {
  id            String        @id @default(cuid())
  subjectId     String
  name          String        // 知识点名称
  code          String?       // 知识点编码
  parentId      String?       // 父级知识点ID(支持多级)
  level         Int           @default(1)  // 层级深度
  order         Int          @default(0)
  
  // 关联
  subject       Subject      @relation(fields: [subjectId], references: [id], onDelete: Cascade)
  parent        Topic?       @relation("TopicHierarchy", fields: [parentId], references: [id])
  children      Topic[]      @relation("TopicHierarchy")
  questionTopics QuestionTopic[]
  
  @@index([subjectId])
  @@index([parentId])
  @@map("topics")
}
```

#### Tag (自定义标签)

```prisma
model Tag {
  id            String          @id @default(cuid())
  userId        String          // 创建者(家长)
  name          String          // 标签名
  color         String          @default("#3B82F6")
  type          TagType         @default(CUSTOM)
  createdAt     DateTime        @default(now())
  
  // 关联
  questionTags  QuestionTag[]
  
  @@unique([userId, name])
  @@map("tags")
}

enum TagType {
  CUSTOM       // 自定义标签
  DIFFICULTY   // 难度标签
  ERROR_TYPE   // 错因标签
}
```

### 2.3 题目系统

#### Question (题目)

```prisma
model Question {
  id            String        @id @default(cuid())
  userId        String        // 创建者
  subjectId     String
  type          QuestionType
  difficulty    Int           @default(3)  // 1-5星难度
  content       Json          // 富文本内容
  answer        Json          // 答案(支持多种格式)
  analysis      Json?         // 题目解析
  isPublic      Boolean       @default(false)  // 是否共享
  version       Int           @default(1)
  status        QuestionStatus @default(DRAFT)
  metadata      Json?         // 扩展字段(来源、作者备注等)
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt
  
  // 关联
  user          User          @relation(fields: [userId], references: [id], onDelete: Cascade)
  subject       Subject       @relation(fields: [subjectId], references: [id])
  versions      QuestionVersion[]
  wrongQuestions WrongQuestion[]
  questionTopics QuestionTopic[]
  questionTags  QuestionTag[]
  attachments   Attachment[]
  recordings    Recording[]
  
  @@index([userId])
  @@index([subjectId])
  @@index([difficulty])
  @@index([type])
  @@map("questions")
}

enum QuestionType {
  SINGLE_CHOICE    // 单选题
  MULTIPLE_CHOICE  // 多选题
  TRUE_FALSE       // 判断题
  FILL_BLANK       // 填空题
  SHORT_ANSWER     // 简答题
  ESSAY            // 论述题
  CALCULATION      // 计算题
  PROOF            // 证明题
  COMPOSITION      // 作文
  READING          // 阅读理解
  LISTENING        // 听力题
}

enum QuestionStatus {
  DRAFT       // 草稿
  PUBLISHED   // 已发布
  ARCHIVED    // 已归档
}

// content 字段结构示例 (富文本)
{
  "blocks": [
    {
      "type": "paragraph",
      "content": "已知函数 $f(x) = x^2 + 2x + 1$，求..."
    },
    {
      "type": "image",
      "url": "/uploads/xxx.png",
      "alt": "配图"
    },
    {
      "type": "formula",
      "latex": "x = \\frac{-b \\pm \\sqrt{b^2-4ac}}{2a}"
    },
    {
      "type": "geometry",
      "svg": "<svg>...</svg>",
      "data": { /* JSXGraph 结构 */ }
    }
  ]
}
```

### 2.4 错题本系统

#### WrongQuestion (错题)

```prisma
model WrongQuestion {
  id            String        @id @default(cuid())
  childId       String        // 孩子账户
  questionId    String
  source        WrongSource   @default(MANUAL)  // 来源
  wrongAnswer   Json?         // 做错时的答案
  mistakeReason String?       // 错因描述
  notes         String?       // 个人备注
  mastered      Boolean       @default(false)  // 是否已掌握
  masteryLevel  Int           @default(0)      // 掌握程度 0-5
  attempts      Int           @default(0)      // 尝试次数
  lastAttempt   DateTime?
  reviewCount   Int           @default(0)     // 复习次数
  lastReview    DateTime?
  nextReview    DateTime?     // 下次复习时间(基于艾宾浩斯)
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt
  
  // 关联
  child         ChildAccount  @relation(fields: [childId], references: [id], onDelete: Cascade)
  question      Question      @relation(fields: [questionId], references: [id])
  reviews       ReviewRecord[]
  
  @@index([childId])
  @@index([questionId])
  @@index([mastered])
  @@index([nextReview])
  @@map("wrong_questions")
}

enum WrongSource {
  MANUAL        // 手动添加
  PRACTICE      // 练习自动录入
  EXAM          // 考试录入
  IMPORT        // 批量导入
}
```

#### ReviewRecord (复习记录)

```prisma
model ReviewRecord {
  id              String          @id @default(cuid())
  wrongQuestionId  String
  childId         String
  reviewed        Boolean         // 是否真正掌握
  confidence      Int?            // 自信心评分 1-5
  notes           String?         // 复习笔记
  aiHelpUsed      Boolean         @default(false)  // 是否使用了AI帮助
  reviewMethod    ReviewMethod?   // 复习方式
  timeSpent       Int?            // 用时(秒)
  reviewedAt      DateTime        @default(now())
  
  // 关联
  wrongQuestion   WrongQuestion   @relation(fields: [wrongQuestionId], references: [id], onDelete: Cascade)
  child           ChildAccount    @relation(fields: [childId], references: [id])
  
  @@index([wrongQuestionId])
  @@index([childId])
  @@index([reviewedAt])
  @@map("review_records")
}

enum ReviewMethod {
  SELF            // 自主复习
  AI_TUTOR        // AI讲解
  PARENT_TEACH    // 家长讲解
  VIDEO           // 视频讲解
  PRACTICE        // 练习巩固
}
```

## 3. 索引设计

### 3.1 复合索引

```sql
-- 错题筛选常用组合
CREATE INDEX idx_wq_filter ON wrong_questions(child_id, mastered, created_at DESC);
CREATE INDEX idx_wq_subject ON wrong_questions(child_id, question_id);

-- 题目筛选
CREATE INDEX idx_q_filter ON questions(user_id, subject_id, difficulty, type);
CREATE INDEX idx_q_public ON questions(subject_id, is_public, difficulty) WHERE is_public = true;

-- 语音评测
CREATE INDEX idx_sr_child_date ON speech_results(child_id, created_at DESC);
CREATE INDEX idx_sr_template ON speech_results(template_id, score);
```

### 3.2 部分索引

```sql
-- 未掌握的错题(高频查询)
CREATE INDEX idx_wq_unmastered ON wrong_questions(child_id, next_review) 
  WHERE mastered = false;

-- 待复习错题
CREATE INDEX idx_wq_review_due ON wrong_questions(child_id, next_review) 
  WHERE next_review IS NOT NULL AND mastered = false;
```

## 4. 性能优化建议

### 4.1 查询优化

```typescript
// 推荐: 使用 select 减少查询字段
const questions = await prisma.question.findMany({
  where: { userId },
  select: {
    id: true,
    subjectId: true,
    type: true,
    difficulty: true,
    createdAt: true,
  },
  take: 20,
  orderBy: { createdAt: 'desc' },
});

// 避免: 不必要的 include
const questions = await prisma.question.findMany({
  where: { userId },
  include: {
    user: true,         // 不需要
    attachments: true,  // 大概率不需要
  },
});
```

### 4.2 N+1 问题解决

```typescript
// 推荐: 使用 findMany + include
const wrongQuestions = await prisma.wrongQuestion.findMany({
  where: { childId },
  include: {
    question: {
      include: {
        subject: true,
        questionTags: {
          include: { tag: true },
        },
      },
    },
  },
});
```
